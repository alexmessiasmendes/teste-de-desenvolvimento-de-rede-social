server.py (FastAPI + WebSockets) 
from fastapi import FastAPI, WebSocket, WebSocketDisconnect, HTTPException
from pydantic import BaseModel
import asyncio

app = FastAPI(title="Live Social Hub Backend")

# In-memory storage for active users and username validation
registered_usernames = set(["casimiro", "virginia", "neymarjr", "marina"])
active_connections: list[WebSocket] = []

class UsernameCheck(BaseModel):
    username: str

@app.post("/api/v1/users/check-username")
async def check_username(data: UsernameCheck):
    """Verifica se o apelido/username já está em uso"""
    clean_name = data.username.strip().lower()
    if clean_name in registered_usernames:
        return {"available": False, "message": "Apelido já em uso."}
    return {"available": True, "message": "Apelido disponível!"}

@app.websocket("/ws/comments")
async def comments_websocket_endpoint(websocket: WebSocket):
    """WebSocket endpoint para streaming de comentários em tempo real"""
    await websocket.accept()
    active_connections.append(websocket)
    try:
        while True:
            # Recebe novo comentário com texto e imagem anexa em JSON
            payload = await websocket.receive_json()
            
            # Broadcast para todos os usuários conectados
            for connection in active_connections:
                await connection.send_json({
                    "event": "NEW_COMMENT",
                    "data": payload
                })
    except WebSocketDisconnect:
        active_connections.remove(websocket)
SocialHub.cs (.NET 8 SignalR Hub) 
using Microsoft.AspNetCore.SignalR;
using System.Collections.Concurrent;

namespace SocialHub.Web.Hubs
{
    public class CommentPayload
    {
        public string PostId { get; set; }
        public string AuthorName { get; set; }
        public string Content { get; set; }
        public string ImageUrl { get; set; }
        public DateTime Timestamp { get; set; } = DateTime.UtcNow;
    }

    public class TrendingSocialHub : Hub
    {
        private static readonly ConcurrentDictionary ActiveUsernames = new();

        public async Task ValidateUsername(string username)
        {
            var cleanName = username.Trim().ToLower();
            if (ActiveUsernames.Values.Contains(cleanName))
            {
                return false; // Apelido ocupado
            }
            ActiveUsernames[Context.ConnectionId] = cleanName;
            return true;
        }

        public async Task BroadcastComment(CommentPayload comment)
        {
            // Dispara evento em tempo real para todos os clientes ativos
            await Clients.All.SendAsync("ReceiveLiveComment", comment);
        }

        public async Task ToggleReaction(string postId, string reactionType)
        {
            await Clients.All.SendAsync("UpdateReactionCount", postId, reactionType);
        }
    }
}
CommentController.java (Spring Boot REST Service) 
package com.socialhub.trending.controller;

import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import java.util.*;

@RestController
@RequestMapping("/api/v1/social")
@CrossOrigin(origins = "*")
public class CommentController {

    private final Set registeredUsers = Collections.synchronizedSet(new HashSet<>());

    @GetMapping("/users/validate/{username}")
    public ResponseEntity> validateUser(@PathVariable String username) {
        String normalized = username.trim().toLowerCase();
        boolean isAvailable = !registeredUsers.contains(normalized);
        
        Map response = new HashMap<>();
        response.put("username", normalized);
        response.put("available", isAvailable);
        
        if (isAvailable) {
            registeredUsers.add(normalized);
        }
        return ResponseEntity.ok(response);
    }
}

// Sincronização ao vivo com Firestore e Reações
import { getFirestore, onSnapshot, collection, addDoc } from "firebase/firestore";

const db = getFirestore();

// Escutar comentários do post em tempo real
function subscribeToComments(postId) {
  const commentsRef = collection(db, 'artifacts', appId, 'public', 'data', `comments_${postId}`);
  
  onSnapshot(commentsRef, (snapshot) => {
    snapshot.docChanges().forEach((change) => {
      if (change.type === "added") {
        renderNewComment(postId, change.doc.data());
      }
    });
  });
}
